if(NOT SVRT_BUILD_PI_LIBRARY OR NOT UNIX)
  return()
endif()

find_package(PkgConfig REQUIRED)
pkg_check_modules(SVRT_AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
pkg_check_modules(SVRT_DRM REQUIRED IMPORTED_TARGET libdrm)

if(SVRT_BUILD_VENDORED_SDL)
  include(FetchContent)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST OFF CACHE BOOL "" FORCE)
  set(SDL_KMSDRM ON CACHE BOOL "" FORCE)
  FetchContent_Declare(SDL2
    GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
    GIT_TAG release-2.32.8)
  FetchContent_MakeAvailable(SDL2)
  set(SVRT_SDL_TARGET SDL2::SDL2-static)
else()
  pkg_check_modules(SVRT_SDL REQUIRED IMPORTED_TARGET sdl2)
  set(SVRT_SDL_TARGET PkgConfig::SVRT_SDL)
endif()

add_library(svrt STATIC svrt.c drm_presenter.c drm_geometry.c)
add_library(svrt::svrt ALIAS svrt)
target_compile_features(svrt PUBLIC c_std_11)
target_include_directories(svrt PUBLIC include PRIVATE .)
target_link_libraries(svrt PUBLIC svrt::pipe PkgConfig::SVRT_AV ${SVRT_SDL_TARGET} PkgConfig::SVRT_DRM)
target_compile_definitions(svrt PRIVATE _POSIX_C_SOURCE=200809L)

install(TARGETS svrt ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(SVRT_BUILD_TESTS)
  add_executable(svrt-geometry-test tests/geometry_test.c drm_geometry.c)
  target_include_directories(svrt-geometry-test PRIVATE .)
  add_test(NAME drm-geometry COMMAND svrt-geometry-test)
endif()
